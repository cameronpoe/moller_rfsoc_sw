3int integrate_one_window(
4 const rotated_sample_t *data,
5 size_t n_samples,
6 gate_window_t window,
7 integration_result_t *out) {
8 /* Validate the input and output pointers before accessing them. */
9 if (!data || !out)
10 return -1;
11 /* A valid integration window must have a positive duration. */
12 if (window.start_ts >= window.end_ts)
13 return -2;
14
15 uint64_t start = window.start_ts;
16 uint64_t end = window.end_ts;
17
18 /* Accumulate the signal sum and count samples that lie inside
19 * the requested half-open timestamp interval [start, end).
20 */
21 double sum = 0.0;
22 size_t count = 0;
23
24 for (size_t i = 0; i < n_samples; i++) {
25 uint64_t ts = data[i].ts;
26
27 /* Include the starting boundary but exclude the ending
28 * boundary to avoid overlap between adjacent windows.
29 */
30 if (ts >= start && ts < end) {
31 sum += data[i].sig;
32 count++;
33 }
34 }
35
36 /* Store the values that are defined even when the window
37 * contains no signal samples.
38 */
39 out->sum = sum;
40 out->n_samples = count;
41
42 /* The arithmetic mean is undefined when no samples were found. */
43 if (count == 0) {
44 out->mean = 0.0;
45 return -3;
46 }
47
48 /* Compute the arithmetic mean of all samples in the window. */
49 out->mean = sum / (double)count;
50 return 0;
51}

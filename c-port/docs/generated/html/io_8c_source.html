10 /* Each sample in the original format consists of exactly
11 * one timestamp word followed by three ADC data words.
12 */
13 if (num_words % 4 != 0)
14 return -2;
15
16 size_t n_packets = num_words / 4;
17
18 /* Parse each four-word sample record independently. */
19 for (size_t samp = 0; samp < n_packets; samp++) {
20 size_t i = 4 * samp;
21
22 /* Read the timestamp and three packed ADC data words. */
23 uint64_t ts_word = raw_data[i];

26 uint64_t word2 = raw_data[i + 3];

32 output[samp].data = adc_frame_parser(word0, word1, word2);

56int adc_packet_parser_v2(
57 const uint64_t *raw_data,
58 size_t num_words,
59 size_t words_per_dma_packet,
60 adc_sample_t *output) {
61 /* Validate the input and output array pointers. */
62 if (!raw_data || !output)
63 return -1;
64
65 /* A valid packet requires one magic word, one timestamp word,
66 * and at least one group of three ADC data words.
67 */
68 if (words_per_dma_packet < 5)
69 return -2;
70
71 /* After removing the two header words, the packet payload must
72 * consist entirely of three-word ADC samples.
73 */
74 if ((words_per_dma_packet - 2) % 3 != 0)
75 return -3;
76
77 /* Reject an input buffer that ends with an incomplete DMA packet. */
78 size_t leftover_words = num_words % words_per_dma_packet;
79 if (leftover_words != 0)
80 return -4;
81
82 size_t num_packets =
83 (num_words - leftover_words) / words_per_dma_packet;
84
85 size_t out_idx = 0;
86
87 /* Process each complete DMA packet in the input buffer. */
88 for (size_t p = 0; p < num_packets; p++) {
89 size_t base = p * words_per_dma_packet;
90
91 /* Verify the synchronization word at the start of the packet.
92 */
93 if (raw_data[base] != MAGIC_WORD)
94 return -5;
95
96 /* Extract the packet timestamp while clearing its
97 * most-significant control bit.
98 */
99 uint64_t packet_ts = raw_data[base + 1] & 0x7FFFFFFFFFFFFFFFULL;
100
101 /* Decode every three-word ADC sample in the packet payload. */
102 size_t samples_per_packet = (words_per_dma_packet - 2) / 3;
103
104 for (size_t s = 0; s < samples_per_packet; s++) {
105 size_t word_idx = base + 2 + 3 * s;
106
107 uint64_t word0 = raw_data[word_idx + 0];
108 uint64_t word1 = raw_data[word_idx + 1];
109 uint64_t word2 = raw_data[word_idx + 2];
110
111 /* Reconstruct the individual sample timestamp from
112 * the packet timestamp and fixed sample interval.
113 */
114 output[out_idx].ts = packet_ts + s * TIME_INC;
115
116 /* Decode the packed ADC words into eight channels. */
117 output[out_idx].data =
118 adc_frame_parser(word0, word1, word2);
119
120 out_idx++;
121 }
122 }

34static inline adc_frame_t adc_frame_parser(
35 uint64_t word1,
36 uint64_t word2,
37 uint64_t word3) {
38 /* Reverse the byte order of each incoming 64-bit word to match
39 * the byte ordering used by the ADC data format.
40 */
41 uint64_t tmp1 = __builtin_bswap64(word1);
42 uint64_t tmp2 = __builtin_bswap64(word2);
43 uint64_t tmp3 = __builtin_bswap64(word3);
44
45 /* Extract the eight individual bytes from the first data word,
46 * beginning with its least-significant byte after byte swapping.
47 */
48 uint8_t b0 = tmp1 & 0xFF;
49 uint8_t b1 = (tmp1 >> 8) & 0xFF;
50 uint8_t b2 = (tmp1 >> 16) & 0xFF;
51 uint8_t b3 = (tmp1 >> 24) & 0xFF;
52 uint8_t b4 = (tmp1 >> 32) & 0xFF;
53 uint8_t b5 = (tmp1 >> 40) & 0xFF;
54 uint8_t b6 = (tmp1 >> 48) & 0xFF;
55 uint8_t b7 = (tmp1 >> 56) & 0xFF;
56
57 /* Extract the eight individual bytes from the second data word. */
58 uint8_t b0_s = tmp2 & 0xFF;
59 uint8_t b1_s = (tmp2 >> 8) & 0xFF;
60 uint8_t b2_s = (tmp2 >> 16) & 0xFF;
61 uint8_t b3_s = (tmp2 >> 24) & 0xFF;
62 uint8_t b4_s = (tmp2 >> 32) & 0xFF;
63 uint8_t b5_s = (tmp2 >> 40) & 0xFF;
64 uint8_t b6_s = (tmp2 >> 48) & 0xFF;
65 uint8_t b7_s = (tmp2 >> 56) & 0xFF;
66
67 /* Extract the eight individual bytes from the third data word. */
68 uint8_t b0_t = tmp3 & 0xFF;
69 uint8_t b1_t = (tmp3 >> 8) & 0xFF;
70 uint8_t b2_t = (tmp3 >> 16) & 0xFF;
71 uint8_t b3_t = (tmp3 >> 24) & 0xFF;
72 uint8_t b4_t = (tmp3 >> 32) & 0xFF;
73 uint8_t b5_t = (tmp3 >> 40) & 0xFF;
74 uint8_t b6_t = (tmp3 >> 48) & 0xFF;
75 uint8_t b7_t = (tmp3 >> 56) & 0xFF;
76
77 /* Reassemble the 24 extracted bytes into eight consecutive
78 * three-byte ADC channel values.
79 */
80 adc_frame_t output;
81
82 output.ch0 = make_adc24(b0, b1, b2);
83 output.ch1 = make_adc24(b3, b4, b5);
84 output.ch2 = make_adc24(b6, b7, b0_s);
85 output.ch3 = make_adc24(b1_s, b2_s, b3_s);
86 output.ch4 = make_adc24(b4_s, b5_s, b6_s);
87 output.ch5 = make_adc24(b7_s, b0_t, b1_t);
88 output.ch6 = make_adc24(b2_t, b3_t, b4_t);
89 output.ch7 = make_adc24(b5_t, b6_t, b7_t);
90
91 return output;
92}

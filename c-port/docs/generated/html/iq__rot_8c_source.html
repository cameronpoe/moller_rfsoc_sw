3int get_iq_data_from_packets_ts(
4 const adc_sample_t *packets,
5 size_t n_samples,
6 int channel,
7 timestamped_iq_t *out) {
8 /* Validate the input and output arrays. */
9 if (!packets || !out)
10 return -1;
11
12 /* Four physical ADC channels are available. Each channel uses
13 * two streams: one for I and one for Q.
14 */
15 if (channel < 0 || channel > 3)
16 return -2;
17
18 /* Extract one complex I/Q value from every timestamped ADC frame. */
19 for (size_t i = 0; i < n_samples; i++) {
20 const adc_frame_t *frame = &packets[i].data;
21
22 /* Select the consecutive I and Q streams belonging to the
23 * requested physical channel.
24 */
25 const adc24_t *ch1_i_adc =
26 get_adc24_channel(frame, 2 * channel);
27 const adc24_t *ch1_q_adc =
28 get_adc24_channel(frame, 2 * channel + 1);
29
30 /* These pointers should always be valid because the channel
31 * range was checked above.
32 */
33 if (!ch1_i_adc || !ch1_q_adc)
34 return -2;
35
36 /* Sign-extend both 24-bit ADC values to signed 32-bit values.
37 */
38 int32_t ch1_i = adc24_to_int32(*ch1_i_adc);
39 int32_t ch1_q = adc24_to_int32(*ch1_q_adc);
40
41 /* Preserve the original sample timestamp and combine the
42 * two components into the complex value I + iQ.
43 */
44 out[i].ts = packets[i].ts;
45 out[i].sig = (double)ch1_i + I * (double)ch1_q;
46 }
47
48 return 0;
49}

60 * initialized to zero so that the caller never observes an

62 */

64 return -1;
65
66 *n_out = 0;

68 * a separate contiguous complex signal array.
69 */

74
75 if (!iq_data || !sig) {
76 free(iq_data);
77 free(sig);
78 return -2;
79 }
80
81 /* Extract the selected physical ADC channel as complex I/Q data. */
82 if (get_iq_data_from_packets_ts(packets, n_samples, channel, iq_data) !=
83 0) {
84 free(iq_data);
85 free(sig);
86 return -3;
87 }
88

90

92 * timestamps, and accumulate the complex signal mean.
93 */
94 for (size_t i = 0; i < n_samples; i++) {
95 sig[i] = iq_data[i].sig;
96 out[i].ts = iq_data[i].ts;
97
98 mean += sig[i];
99 }
100
101 /* Determine whether the extracted signal is already sufficiently
102 * constant in amplitude and phase to be treated as DC.
103 */
104 mean /= (double)n_samples;
105
106 bool sig_dc = check_if_dc(sig, n_samples, PHASE_LIM, REL_LIM);
107 if (!sig_dc) {
108 /* FFT down-conversion requires a nonzero power-of-two
109 * block length.
110 */
111 if (fft_len == 0 || (fft_len & (fft_len - 1)) != 0) {
112 free(iq_data);
113 free(sig);
114 return -4;
115 }
116
117 /* At least one complete FFT block must fit in the signal. */
118 if (fft_len > n_samples) {
119 free(iq_data);
120 free(sig);
121 return -5;
122 }
123
124 /* Allocate storage for the real-valued down-converted signal.
125 */
126 double *real = (double *)calloc(n_samples, sizeof(double));
127
128 if (!real) {

130 free(iq_data);
131 free(sig);
132 return -6;
133 }
134

136 size_t usable = n_samples - (n_samples % fft_len);
137
138 /* Down-convert each complete FFT block independently. */
139 for (size_t start = 0; start + fft_len <= usable;

141 int s = process_fft_block(
142 sig, start, fft_len, (double)samp_freq, real);
143
144 if (s != 0) {
145 free(real);
146 free(iq_data);
147 free(sig);
148 return -7;
149 }
150 }
151
152 /* Copy all successfully processed FFT samples into the output
153 * array. Only complete FFT blocks are processed, therefore any
154 * incomplete trailing block is discarded.
155 */
156 for (size_t i = 0; i < usable; i++) {
157 out[i].ts = iq_data[i].ts;
158 out[i].sig = real[i];
159 }
160
161 /* Report the number of valid output samples. */
162 *n_out = usable;
163
164 free(real);
165 free(iq_data);
166 free(sig);
167
168 return 0;
169 }
170
171 /* The signal is already at DC. Remove its residual constant phase
172 * so that the complex mean lies on the positive real axis.
173 */
174 double complex rot = cexp(-I * carg(mean));
175
176 /* Rotate the signal so that its mean lies on the positive real axis,
177 * then retain only the real component.
178 */
179 for (size_t i = 0; i < n_samples; i++) {
180 out[i].sig = creal(sig[i] * rot);
181 }
182
183 /* Every input sample produced one output sample. */
184 *n_out = n_samples;
185 free(iq_data);
186 free(sig);
187 return 0;

203
204 /* A valid timestamp window must have a positive duration. */

217 uint64_t ts = packets[i].ts;
218

220 count++;
221 }
222 }
223
224 /* The selected window cannot be processed if it contains no samples. */
225 if (count == 0)
226 return -3;
227
228 /* Allocate a contiguous array containing only samples from the
229 * requested timestamp window.
230 */
231 adc_sample_t *window_packets =
232 (adc_sample_t *)calloc(count, sizeof(adc_sample_t));
233
234 if (!window_packets)
235 return -4;
236

238 size_t j = 0;
239

241 uint64_t ts = packets[i].ts;

244 window_packets[j] = packets[i];
245 j++;

247 }

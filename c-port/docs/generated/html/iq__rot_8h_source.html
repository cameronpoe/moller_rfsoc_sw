42static inline bool check_if_dc(
43 double complex *sig,
44 size_t n,
45 double phase_lim,
46 double rel_lim) {
47
48 /* At least two samples are required to calculate a
49 * sample-to-sample difference.
50 */
51 if (!sig || n < 2)
52 return false;
53
54 double sum_abs_dphi = 0.0;
55 double sum_amp = 0.0;
56 double accum = 0.0;
57
58 /* Accumulate the amplitude, sample-to-sample change, and phase
59 * difference for every consecutive pair of signal samples.
60 */
61 for (size_t i = 0; i + 1 < n; i++) {
62 double complex diff = sig[i + 1] - sig[i];
63
64 sum_amp += cabs(sig[i]);
65 accum += cabs(diff);
66
67 /* x[i + 1] * conj(x[i]) has phase equal to the phase
68 * difference between the two consecutive samples.
69 */
70 double complex phase_ratio = sig[i + 1] * conj(sig[i]);
71 sum_abs_dphi += fabs(carg(phase_ratio));
72 }
73
74 /* Compute averages over the n - 1 consecutive sample pairs. */
75 double mean_amp = sum_amp / (double)(n - 1);
76 double mean_diff = accum / (double)(n - 1);
77 double mean_phase_diff = sum_abs_dphi / (double)(n - 1);
78
79 /* The relative variation is undefined for a zero-amplitude signal. */
80 if (mean_amp == 0)
81 return false;
82
83 double rel_diff = mean_diff / mean_amp;
84
85 /* Classify the signal as DC only when both amplitude and phase
86 * variation remain below their respective limits.
87 */
88 return (rel_diff < rel_lim) && (mean_phase_diff < phase_lim);
89}

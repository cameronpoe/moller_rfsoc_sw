235static inline int fft_Cooley_Tukey(
236 double complex *arr,
237 size_t fft_len,
238 double complex *out) {
239 /* Validate input pointers. */
240 if (!arr || !out)
241 return -1;
242 /* The Cooley-Tukey algorithm requires the input length to be non-zero
243 * and a powewr of two. */
244 if (fft_len == 0 || (fft_len & (fft_len - 1)) != 0)
245 return -2;
246 /* Base case of recursion. The FFT of a single elemet is an element
247 * itself. */
248 if (fft_len == 1) {
249 out[0] = arr[0];
250 return 0;
251 }
252
253 size_t half = fft_len / 2;
254
255 /* Allocate temporary buffers for the even and odd subsequences
256 * together with their corresponding FFT outputs.
257 */
258 double complex *even =
259 (double complex *)calloc(half, sizeof(double complex));
260 double complex *odd =
261 (double complex *)calloc(half, sizeof(double complex));
262 double complex *even_fft =
263 (double complex *)calloc(half, sizeof(double complex));
264 double complex *odd_fft =
265 (double complex *)calloc(half, sizeof(double complex));
266
267 /* Abort if memory allocation failed. */
268 if (!odd || !even || !even_fft || !odd_fft) {
269 free(odd);
270 free(even);
271 free(even_fft);
272 free(odd_fft);
273 return -3;
274 }
275
276 /* Split the input sequence into its even- and odd-indexed
277 * subsequences.
278 */
279 int status1 = get_odd(arr, fft_len, odd);
280 int status2 = get_even(arr, fft_len, even);
281
282 if ((status1 != 0) || (status2 != 0)) {
283 free(odd);
284 free(even);
285 free(even_fft);
286 free(odd_fft);
287 return -4;
288 }
289
290 /* Recursively compute the FFT of the two subsequences. */
291 int status3 = fft_Cooley_Tukey(even, half, even_fft);
292 int status4 = fft_Cooley_Tukey(odd, half, odd_fft);
293
294 if (status3 != 0 || status4 != 0) {
295 free(even);
296 free(odd);
297 free(even_fft);
298 free(odd_fft);
299 return -5;
300 }
301
302 /* Combine the two half-size FFTs into the FFT of the full
303 * sequence using the Cooley-Tukey butterfly operation.
304 */
305 for (size_t k = 0; k < half; k++) {
306 double angle = -2.0 * M_PI * k / fft_len;
307 double complex w = cos(angle) + I * sin(angle);
308
309 out[k] = even_fft[k] + w * odd_fft[k];
310 out[k + half] = even_fft[k] - w * odd_fft[k];
311 }
312
313 /*Releases all temporay buffers.*/
314 free(odd);
315 free(even);
316 free(even_fft);
317 free(odd_fft);
318
319 return 0;
320}

337static inline double fft_bin_to_freq(
338 double sampl_freq,
339 size_t n,
340 size_t k) {
341 if (k <= n / 2) {
342 return (double)k * sampl_freq / (double)n;

344 return -((double)(n - k) * sampl_freq) / (double)n;
345 }

365static inline size_t fft_find_argmax(
366 double complex *out,
367 size_t n) {
368 double max = cabs(out[0]);
369 size_t best_ind = 0;
370
371 if (!out || n == 0)
372 return 0;
373
374 for (size_t i = 1; i < n; i++) {
375 double tmp = cabs(out[i]);
376
377 if (tmp > max) {
378 max = tmp;
379 best_ind = i;
380 }
381 }
382
383 return best_ind;
384}

414static inline int process_fft_block(
415 const double complex *sig,
416 size_t start,
417 size_t fft_len,
418 double samp_freq,
419 double *out) {
420 /* Validate the input and output pointers before accessing
421 * either array.
422 */
423 if (!sig || !out)
424 return -1;
425 /* Allocate temporary buffers for the FFT coefficients and the
426 * complex signal after down-conversion to DC.
427 */
428 double complex *fft_out = calloc(fft_len, sizeof(double complex));
429 double complex *dc = calloc(fft_len, sizeof(double complex));
430
431 /* Release any successfully allocated buffer if either
432 * allocation failed.
433 */
434 if (!fft_out || !dc) {
435 free(fft_out);
436 free(dc);
437 return -2;
438 }
439
440 /* Compute the FFT of the selected input block.
441 *
442 * The cast can be removed if fft_Cooley_Tukey() accepts
443 * const double complex
444 * as its input argument.
445 */
446 int status = fft_Cooley_Tukey(
447 (double complex *)&sig[start], fft_len, fft_out);
448
449 /* Abort if the FFT calculation failed. */
450 if (status != 0) {
451 free(fft_out);
452 free(dc);
453 return -3;
454 }
455
456 /* Find the FFT bin containing the largest signal magnitude. */
457 size_t kmax = fft_find_argmax(fft_out, fft_len);
458
459 /* SIZE_MAX indicates that the FFT output was invalid or empty. */
460 if (kmax == SIZE_MAX) {
461 free(fft_out);
462 free(dc);
463 return -4;
464 }
465
466 /* Convert the dominant FFT-bin index to its signed physical
467 * frequency.
468 */
469 double freq = fft_bin_to_freq(samp_freq, fft_len, kmax);
470 /* Extract the phase of the dominant carrier component.
471 *
472 * This value is currently used only for diagnostic output.
473 */
474 double carrier_phase = carg(fft_out[kmax]);
475
476 printf("C DEBUG: start=%zu fft_len=%zu kmax=%zu carrier_freq=%.17g "
477 "carrier_phase=%.17g\n",
478 start,
479 fft_len,
480 kmax,
481 freq,
482 carrier_phase);
483 /* Mix the dominant carrier frequency down to DC.
484 *
485 * Multiplication by exp(-i * phase) removes the estimated
486 * carrier oscillation from each complex input sample.
487 */
488 for (size_t i = 0; i < fft_len; i++) {
489 double phase = 2.0 * M_PI * freq * (double)i / samp_freq;
490 dc[i] = sig[start + i] * cexp(-I * phase);
491 }
492
493 /* Compute the mean complex value of the down-converted block. */
494 double complex mean = 0.0 + 0.0 * I;
495 for (size_t i = 0; i < fft_len; i++) {
496 mean += dc[i];
497 }
498 mean /= (double)fft_len;
499
500 /* Remove the residual phase offset so that the mean signal
501 * lies on the positive real axis.
502 */
503 double complex rot = cexp(-I * carg(mean));
504
505 /* Apply the phase rotation and store only the real component
506 * of each final DC sample.
507 */
508 for (size_t j = 0; j < fft_len; j++) {
509 out[start + j] = creal(dc[j] * rot);
510 }
511
512 /* Release all temporary working buffers. */
513 free(fft_out);
514 free(dc);
515
516 return 0;
517}

557static inline int32_t adc24_to_int32(
558 adc24_t x) {
559 /* Combine the three bytes into one unsigned 24-bit value.
560 *
561 * b0 becomes the most-significant byte and b2 becomes the
562 * least-significant byte.
563 */
564 uint32_t raw = ((uint32_t)x.b0 << 16) | ((uint32_t)x.b1 << 8) |
565 ((uint32_t)x.b2);
566
567 /* Move the original sign bit from bit 23 to bit 31, convert
568 * to a signed integer, and shift it back with sign extension.
569 */
570 return ((int32_t)(raw << 8)) >> 8;
571}
